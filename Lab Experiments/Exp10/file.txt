import heapq
from typing import Any, Callable, Dict, List, Tuple

def astar(
    start: Any,
    goal: Any,
    neighbors_fn: Callable[[Any], List[Tuple[Any, float]]],
    heuristic_fn: Callable[[Any, Any], float]
) -> Tuple[List[Any], float]:
  
    frontier = []
    heapq.heappush(frontier, (0, start))
    came_from: Dict[Any, Any] = {start: None}
    cost_so_far: Dict[Any, float] = {start: 0}

    while frontier:
        _, current = heapq.heappop(frontier)

        if current == goal:
            break

        for neighbor, edge_cost in neighbors_fn(current):
            new_cost = cost_so_far[current] + edge_cost
            if neighbor not in cost_so_far or new_cost < cost_so_far[neighbor]:
                cost_so_far[neighbor] = new_cost
                priority = new_cost + heuristic_fn(neighbor, goal)
                heapq.heappush(frontier, (priority, neighbor))
                came_from[neighbor] = current

    # If goal not reached
    if goal not in came_from:
        return [], float("inf")

    # Reconstruct path
    path = []
    current = goal
    while current is not None:
        path.append(current)
        current = came_from[current]
    path.reverse()

    return path, cost_so_far[goal]


# Example usage
if __name__ == "__main__":
    # Graph as adjacency list with weights
    graph = {
        'A': [('B', 1), ('C', 3)],
        'B': [('D', 1), ('E', 5)],
        'C': [('F', 2)],
        'D': [],
        'E': [('F', 1)],
        'F': []
    }

    def neighbors_fn(node):
        return graph.get(node, [])

    def heuristic_fn(node, goal):
        # Simple heuristic: zero (Dijkstra’s behavior)
        return 0

    path, cost = astar('A', 'F', neighbors_fn, heuristic_fn)
    print("Path:", path)
    print("Cost:", cost)
