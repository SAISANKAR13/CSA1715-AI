bird(sparrow). bird(eagle). bird(crow). bird(parrot). bird(penguin). bird(ostrich). bird(kiwi).   
cannot_fly(penguin). cannot_fly(ostrich). cannot_fly(kiwi).   

can_fly(B) :- bird(B), \+ cannot_fly(B).   

check(B) :-     
    (   can_fly(B)  -> format('~w can fly~n', [B])     
    ;   bird(B)     -> format('~w cannot fly~n', [B])     
    ;   format('~w is not in the knowledge base~n', [B])     
    ).
